<!DOCTYPE html>
<html>
<head>
  <title>CRUD</title>
</head>
<body>
  <h2>Enter something</h2>
  <input type="text" id="text">
  <button onclick="save()">Save</button>

  <h3>Saved items</h3>
  <ul id="list"></ul>

  <script>
    // CREATE
    async function save() {
      const input = document.getElementById('text');
      if (!input.value) return;
      await fetch('/api/entries', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ text: input.value })
      });
      input.value = '';
      load();
    }

    // UPDATE
    async function edit(id, oldText) {
      const newText = prompt('Edit:', oldText);
      if (!newText) return;
      await fetch('/api/entries/' + id, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ text: newText })
      });
      load();
    }

    // DELETE
    async function del(id) {
      await fetch('/api/entries/' + id, { method: 'DELETE' });
      load();
    }

    // READ
    async function load() {
      const res = await fetch('/api/entries');
      const data = await res.json();
      const list = document.getElementById('list');
      list.innerHTML = '';
      data.forEach(e => {
        const li = document.createElement('li');
        li.textContent = e.text + ' ';

        const editBtn = document.createElement('button');
        editBtn.textContent = 'Edit';
        editBtn.onclick = () => edit(e._id, e.text);

        const delBtn = document.createElement('button');
        delBtn.textContent = 'Delete';
        delBtn.onclick = () => del(e._id);

        li.append(editBtn, delBtn);
        list.appendChild(li);
      });
    }

    load();
  </script>
</body>
</html>

require('dotenv').config();
const express = require('express');
const mongoose = require('mongoose');

const app = express();
app.use(express.json());
app.use(express.static('public'));

mongoose.connect(process.env.MONGODB_URI)
  .then(() => console.log('MongoDB connected'))
  .catch((err) => console.error('MongoDB connection error:', err));

const Entry = mongoose.model('Entry',
  new mongoose.Schema({ text: String }, { timestamps: true }));

// CREATE
app.post('/api/entries', async (req, res) => {
  res.json(await Entry.create({ text: req.body.text }));
});

// READ
app.get('/api/entries', async (req, res) => {
  res.json(await Entry.find().sort({ createdAt: -1 }));
});

// UPDATE
app.put('/api/entries/:id', async (req, res) => {
  res.json(await Entry.findByIdAndUpdate(
    req.params.id, { text: req.body.text }, { new: true }));
});

// DELETE
app.delete('/api/entries/:id', async (req, res) => {
  await Entry.findByIdAndDelete(req.params.id);
  res.json({ message: 'deleted' });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
