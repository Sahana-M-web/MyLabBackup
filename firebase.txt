in firebase console add project register it enable auth and create db and go to firestore rules add this 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /notes/{noteId} {
      allow read, delete: if request.auth != null && resource.data.uid == request.auth.uid;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
    }
  }
}

    getAuth, onAuthStateChanged, createUserWithEmailAndPassword,
    signInWithEmailAndPassword, signInWithPopup, GoogleAuthProvider, signOut
  } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js";
  import {
    getFirestore, collection, addDoc, deleteDoc, doc, query, where,
    orderBy, onSnapshot, serverTimestamp
  } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js";

const firebaseConfig = {
  apiKey: "AIzaSyD0ukEFxAdO2ISxiBUZc648ZKn1ucPULHw",
  authDomain: "sahana-ram.firebaseapp.com",
  projectId: "sahana-ram",
  storageBucket: "sahana-ram.firebasestorage.app",
  messagingSenderId: "958636582430",
  appId: "1:958636582430:web:4bfa5b7ae7010b773eb00f",
  measurementId: "G-ZMDN565ZBF"
};


  const app  = initializeApp(firebaseConfig);
  const auth = getAuth(app);
  const db   = getFirestore(app);

  const $ = (id) => document.getElementById(id);
  const showError = (e) => { $("msg").textContent = e.message.replace("Firebase: ", ""); };
  let unsubscribeNotes = null;

  // 3. AUTHENTICATION
  $("signupBtn").onclick = () =>
    createUserWithEmailAndPassword(auth, $("email").value, $("password").value).catch(showError);

  $("loginBtn").onclick = () =>
    signInWithEmailAndPassword(auth, $("email").value, $("password").value).catch(showError);

  $("googleBtn").onclick = () =>
    signInWithPopup(auth, new GoogleAuthProvider()).catch(showError);

  $("logoutBtn").onclick = () => signOut(auth);

  // Runs on page load and whenever the user signs in or out
  onAuthStateChanged(auth, (user) => {
    $("authView").classList.toggle("hidden", !!user);
    $("appView").classList.toggle("hidden", !user);
    $("msg").textContent = "";

    if (unsubscribeNotes) { unsubscribeNotes(); unsubscribeNotes = null; }

    if (user) {
      $("who").textContent = user.email;
      listenToNotes(user.uid);
    }
  });

  // 4. FIRESTORE: CREATE
  $("addBtn").onclick = async () => {
    const text = $("noteText").value.trim();
    if (!text || !auth.currentUser) return;
    await addDoc(collection(db, "notes"), {
      uid: auth.currentUser.uid,
      text,
      createdAt: serverTimestamp()
    });
    $("noteText").value = "";
  };

  // FIRESTORE: READ (real-time, only the signed-in user's notes)
  // First run: Firestore will log a link in the console to create the index
  // needed for where() + orderBy(). Click it once.
  function listenToNotes(uid) {
    const q = query(collection(db, "notes"), where("uid", "==", uid), orderBy("createdAt", "desc"));
    unsubscribeNotes = onSnapshot(q, (snap) => {
      const list = $("notesList");
      if (snap.empty) { list.textContent = "No notes yet. Add your first one above."; return; }
      list.innerHTML = "";
      snap.forEach((d) => {
        const row = document.createElement("div");
        row.className = "note";
        const span = document.createElement("span");
        span.textContent = d.data().text;           // textContent avoids XSS
        const del = document.createElement("button");
        del.className = "del";
        del.textContent = "Delete";
        // FIRESTORE: DELETE
        del.onclick = () => deleteDoc(doc(db, "notes", d.id));
        row.append(span, del);
        list.append(row);
      });
    }, showError);
  }
