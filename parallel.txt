#include <stdio.h>
#include <omp.h>

int main() {
    int sum = 0;

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= 100; i++) {
        sum += i;
    }

    printf("Sum = %d\n", sum);
    return 0;
}

from mpi4py import MPI

comm = MPI.COMM_WORLD

rank = comm.Get_rank()
size = comm.Get_size()

print("Hello from process", rank,
      "out of", size, "processes")


mpirun --allow-run-as-root -np 4 python3 mpi_hello.py
